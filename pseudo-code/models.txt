classe GameState:
    """
    GameState

    Champs (types attendus) :
    - secret_word : str
    - guessed_letters : set[str]     # lettres correctes déjà trouvées
    - wrong_letters : set[str]       # lettres fausses déjà tentées
    - max_errors : int              # erreurs autorisées selon difficulté
    - status : str                  # "playing" / "won" / "lost"

    Invariants :
    - guessed_letters et wrong_letters ne se chevauchent jamais
    - len(wrong_letters) <= max_errors tant que playing
    """

    fonction __init__(secret_word, guessed_letters, wrong_letters, max_errors, status):
        enregistrer tous les champs
